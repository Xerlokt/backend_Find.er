Create `src/shared/errors/errorHandler.ts`:
```typescript
import { FastifyError, FastifyReply, FastifyRequest } from 'fastify'
import { ZodError } from 'zod'
import { Prisma } from '@prisma/client'
import { AppError } from './AppError'

export function errorHandler(
  error: FastifyError | AppError | Error,
  _request: FastifyRequest,
  reply: FastifyReply,
): void {
  // AppError — domain errors with explicit status
  if (error instanceof AppError) {
    reply.status(error.statusCode).send({ error: error.message })
    return
  }

  // Zod validation errors — SEC-02: input validation failures
  if (error instanceof ZodError) {
    reply.status(400).send({
      error: 'Validation error',
      details: error.errors.map((e) => ({
        field: e.path.join('.'),
        message: e.message,
      })),
    })
    return
  }

  // Prisma known errors — map to meaningful HTTP codes
  if (error instanceof Prisma.PrismaClientKnownRequestError) {
    if (error.code === 'P2002') {
      reply.status(409).send({ error: 'Resource already exists' })
      return
    }
    if (error.code === 'P2025') {
      reply.status(404).send({ error: 'Resource not found' })
      return
    }
  }

  // Fastify validation errors (schema-level)
  if ('statusCode' in error && typeof error.statusCode === 'number') {
    reply.status(error.statusCode).send({ error: error.message })
    return
  }

  // Unknown — log internally, never leak stack to client
  console.error('[errorHandler] Unhandled error:', error)
  reply.status(500).send({ error: 'Internal server error' })
}
```

```typescript
import neo4j, { Driver } from 'neo4j-driver'

let driver: Driver

export function getNeo4jDriver(): Driver {
  if (!driver) {
    driver = neo4j.driver(
      process.env.NEO4J_URI!,
      neo4j.auth.basic(process.env.NEO4J_USER!, process.env.NEO4J_PASSWORD!),
    )
  }
  return driver
}

export async function closeNeo4j(): Promise<void> {
  if (driver) {
    await driver.close()
  }
}
```

```typescript
import { FastifyRequest, FastifyReply } from 'fastify'

    user: {

  _reply: FastifyReply,
```

Create `src/shared/hooks/requireRole.ts`. This higher-order function returns a preHandler that enforces role access. It MUST be used after `authenticate` (which populates request.user):

```typescript
import { FastifyRequest, FastifyReply } from 'fastify'
import { AppError } from '../errors/AppError'

type Role = 'CANDIDATE' | 'RECRUITER' | 'ADMIN'

export function requireRole(...roles: Role[]) {
  return async function roleGuard(
    request: FastifyRequest,
    _reply: FastifyReply,
  ): Promise<void> {
    if (!request.user) {
      throw new AppError('Unauthorized', 401, 'UNAUTHORIZED')
    }
    if (!roles.includes(request.user.role)) {
      throw new AppError('Forbidden', 403, 'FORBIDDEN')
    }
  }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| JWT token → authenticate hook | Token arrives in Authorization header; hook verifies signature and populates request.user |
| errorHandler → HTTP response | Error internals must not leak to client (no stack traces, no DB connection strings) |
| health route → Prisma | Health check exposes DB connectivity status — acceptable at /health; no sensitive data |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-03-01 | Spoofing | src/shared/hooks/authenticate.ts | mitigate | JWT verified via request.jwtVerify() — signature validated against JWT_SECRET; invalid tokens throw 401 AppError |
| T-01-03-02 | Elevation of Privilege | src/shared/hooks/requireRole.ts | mitigate | Role checked against request.user.role (set by authenticate); requireRole() throws 403 if role not in allowed list; hooks are composable but authenticate MUST run first |
| T-01-03-03 | Information Disclosure | src/shared/errors/errorHandler.ts | mitigate | Unknown errors log internally via console.error but send only "Internal server error" to client — no stack trace, no error message propagation |
| T-01-03-04 | Information Disclosure | src/modules/health/health.routes.ts | accept | /health exposes db connectivity boolean — acceptable for internal monitoring; no PII or secrets returned |
| T-01-03-05 | Denial of Service | GET /health (Prisma ping) | accept | $queryRaw SELECT 1 is minimal overhead; rate limiting on sensitive routes added in Phase 2 (SEC-05); health endpoint not rate-limited by design |
</threat_model>
