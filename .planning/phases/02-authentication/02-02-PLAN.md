async refresh(rawToken: string): Promise<TokenPair> {
  const tokenHash = hashToken(rawToken)
  const stored = await this.repo.findRefreshTokenByHash(tokenHash)

  if (!stored) {
    throw new AppError('Invalid token', 401, 'UNAUTHORIZED')
  }

  if (stored.revokedAt !== null) {
    await this.repo.revokeRefreshTokenFamily(stored.familyId)
    throw new AppError('Token revoked', 401, 'TOKEN_EXPIRED')
  }

  if (stored.expiresAt < new Date()) {
    throw new AppError('Token expired', 401, 'TOKEN_EXPIRED')
  }

  await this.repo.revokeRefreshToken(stored.id)

  const user = await this.repo.findUserById(stored.userId)
  if (!user) {
    throw new AppError('User not found', 401, 'UNAUTHORIZED')
  }

  return this.generateTokenPair(user.id, user.role as string, stored.familyId)
}

async logout(rawToken: string): Promise<void> {

  const stored = await this.repo.findRefreshTokenByHash(tokenHash)

  if (!stored || stored.revokedAt !== null) {
    // Token inexistente ou já revogado — idempotente, não lançar erro

  }

  await this.repo.revokeRefreshToken(stored.id)
}

**1. Criar src/modules/auth/auth.routes.ts:**

```typescript
import { FastifyInstance } from 'fastify'
import { z } from 'zod'
import { AuthRepository } from './auth.repository'
import { AuthService } from './auth.service'
import { authenticate } from '../../shared/hooks/authenticate'

const RegisterBodySchema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
  role: z.enum(['CANDIDATE', 'RECRUITER']),
})

const LoginBodySchema = z.object({
  email: z.string().email(),
  password: z.string().min(1),
})

const RefreshBodySchema = z.object({
  refreshToken: z.string().min(1),
})

export async function authRoutes(app: FastifyInstance): Promise<void> {
  const repo = new AuthRepository()
  const service = new AuthService(repo, app)

  // POST /auth/register — 3 req/min (D-10)
  app.post('/register', {
    config: { rateLimit: { max: 3, timeWindow: '1 minute' } },
  }, async (request, reply) => {
    const body = RegisterBodySchema.parse(request.body)
    const tokens = await service.register(body)
    return reply.status(201).send(tokens)
  })

  // POST /auth/login — 5 req/min (D-10)
  app.post('/login', {
    config: { rateLimit: { max: 5, timeWindow: '1 minute' } },
  }, async (request, reply) => {
    const body = LoginBodySchema.parse(request.body)
    const tokens = await service.login(body)
    return reply.status(200).send(tokens)
  })

  // POST /auth/refresh — 10 req/min (D-10)
  // Recebe refreshToken no body — NÃO usa authenticate preHandler (token opaco, não JWT)
  app.post('/refresh', {
    config: { rateLimit: { max: 10, timeWindow: '1 minute' } },
  }, async (request, reply) => {
    const body = RefreshBodySchema.parse(request.body)
    const tokens = await service.refresh(body.refreshToken)
    return reply.status(200).send(tokens)
  })

  // POST /auth/logout — sem rate limit customizado (60 req/min global)
  // Recebe refreshToken no body — NÃO usa authenticate preHandler
  app.post('/logout', async (request, reply) => {
    const body = RefreshBodySchema.parse(request.body)
    await service.logout(body.refreshToken)
    return reply.status(204).send()
  })
}
```

**Notas críticas:**
- `/auth/refresh` e `/auth/logout` recebem o refreshToken opaco no body — NÃO são JWTs, portanto NÃO usam `authenticate` preHandler (RESEARCH.md anti-patterns).
- `/auth/register`, `/auth/login`, `/auth/google` são rotas públicas — sem `authenticate`.
- Rate limits via `config.rateLimit` exigem que o plugin foi registrado com `fp` wrapper (já feito em 02-01).
