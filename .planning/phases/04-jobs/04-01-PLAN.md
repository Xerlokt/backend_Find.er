    STEP 5 — Create src/modules/job/job.service.ts:
    ```typescript
    import { JobRepository, CreateJobData, UpdateJobData } from './job.repository'
    import { AppError } from '../../shared/errors/AppError'

    export class JobService {
      constructor(private readonly repo: JobRepository) {}

      async createJob(userId: string, data: CreateJobData) {
        const profile = await this.repo.findRecruiterProfileByUserId(userId)
        if (!profile) throw new AppError('Recruiter profile not found', 404, 'NOT_FOUND')
        return this.repo.create(profile.id, data)
      }

      async updateJob(userId: string, jobId: string, data: UpdateJobData) {
        const profile = await this.repo.findRecruiterProfileByUserId(userId)
        if (!profile) throw new AppError('Recruiter profile not found', 404, 'NOT_FOUND')
        const job = await this.repo.findJobById(jobId)
        if (!job) throw new AppError('Job not found', 404, 'NOT_FOUND')
        if (job.recruiterId !== profile.id) throw new AppError('Forbidden', 403, 'FORBIDDEN')
        return this.repo.update(jobId, data)
      }

      async deactivateJob(userId: string, jobId: string) {
        return this.updateJob(userId, jobId, { isActive: false })
      }

      async listJobsByRecruiter(userId: string) {
        const profile = await this.repo.findRecruiterProfileByUserId(userId)
        if (!profile) throw new AppError('Recruiter profile not found', 404, 'NOT_FOUND')
        return this.repo.findByRecruiterId(profile.id)
      }

      async getJobById(jobId: string) {
        const job = await this.repo.findJobById(jobId)
        if (!job) throw new AppError('Job not found', 404, 'NOT_FOUND')
        return job
      }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → job routes | Untrusted user input: title, description, skills, salary range, coordinates |
| JWT payload → service | userId and role extracted from verified JWT — trusted after authenticate hook |
| Service → repository | Internal boundary — service must enforce ownership before any repo mutation |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01-01 | Spoofing | job.service.ts createJob | mitigate | recruiterId derived from repo.findRecruiterProfileByUserId(userId) — userId always from JWT, never from request body |
| T-04-01-02 | Tampering | job.service.ts updateJob/deactivateJob | mitigate | IDOR guard: `job.recruiterId !== profile.id` throws AppError 403 before any mutation — covered by unit and integration tests |
| T-04-01-03 | Elevation of Privilege | job.routes.ts (04-03) | mitigate | requireRole('RECRUITER') preHandler blocks CANDIDATE and ADMIN roles — ASVS L1 access control |
| T-04-01-04 | Information Disclosure | job.repository.ts | accept | Job listings filtered by recruiterId — recruiter cannot enumerate other recruiters' jobs via listByRecruiterId |
| T-04-01-05 | Denial of Service | Prisma migration | accept | Migration runs as internal CLI operation with no external input surface; no user data involved |
</threat_model>
