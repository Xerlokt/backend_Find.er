    STEP 1 — Create src/shared/plugins/scheduler.ts:
    ```typescript
    import fp from 'fastify-plugin'
    import { FastifyInstance } from 'fastify'
    import cron from 'node-cron'
    import { JobRepository } from '../../modules/job/job.repository'
    import { runExpiryCheck } from '../../modules/job/job.scheduler'

    async function schedulerPlugin(app: FastifyInstance): Promise<void> {
      const repo = new JobRepository()

      const task = cron.schedule('* * * * *', async () => {
        try {
          const count = await runExpiryCheck(repo)
          if (count > 0) {
            app.log.info({ deactivated: count }, 'Job expiry check: deactivated expired jobs')
          }
        } catch (err) {
          app.log.error({ err }, 'Job expiry check failed')
        }
      }, { scheduled: false })

      task.start()

      app.addHook('onClose', async () => {
        task.stop()
      })
    }

    export const schedulerPlugin = fp(schedulerPlugin)
    ```

    NOTE on node-cron v4: The `schedule()` function returns a ScheduledTask with `.start()` and `.stop()` methods. The `{ scheduled: false }` option prevents auto-start so we call `.start()` explicitly after the hook is registered.

    ```typescript
    import '../helpers/setup'
    import { buildApp } from '../../src/app'

    describe('Scheduler — testing mode guard', () => {
      it('does not register schedulerPlugin when opts.testing=true', async () => {
        const app = await buildApp({ testing: true })
        // If the scheduler were registered, it would create a live cron task.
        // We verify the app has no 'schedulerPlugin' decoration, which would only
        // exist if the plugin ran.
        // The simplest assertion: app closes cleanly with no open handles.
        await app.close()
        // If cron leaked, Jest --detectOpenHandles would report it after this suite.
        // Assert no error was thrown during build + close.
        expect(true).toBe(true)
      })

      it('runExpiryCheck is never auto-triggered in testing mode', async () => {
        // Build app with testing=true, wait a short period, confirm no background expiry
        // runs have occurred (scheduler is entirely absent — no cron timer is scheduled).
        const app = await buildApp({ testing: true })
        // No way for cron to fire in 0ms because it was never started.
        // Verify by checking app does not have the scheduler decoration.
        const hasScheduler = app.hasPlugin('schedulerPlugin')
        // hasPlugin returns false when the plugin was not registered.
        // Note: fastify-plugin wraps with fp() which uses the function name as plugin name.
        expect(hasScheduler).toBe(false)
        await app.close()
      })
    })
    ```
