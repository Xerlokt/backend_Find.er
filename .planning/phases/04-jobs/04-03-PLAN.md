    ```typescript
    import { FastifyInstance } from 'fastify'
    import { z } from 'zod'
    import { authenticate } from '../../shared/hooks/authenticate'
    import { requireRole } from '../../shared/hooks/requireRole'
    import { JobRepository } from './job.repository'
    import { JobService } from './job.service'

    const CreateJobSchema = z.object({
      title: z.string().min(1).max(200),
      description: z.string().min(1).max(5000),
      professionId: z.string().min(1),
      companyId: z.string().uuid(),
      salaryMin: z.number().positive().optional(),
      salaryMax: z.number().positive().optional(),
      lat: z.number().min(-90).max(90).optional(),
      lng: z.number().min(-180).max(180).optional(),
      modality: z.enum(['REMOTE', 'HYBRID', 'IN_PERSON']),
      requiredSkills: z.array(z.string()).min(1),
      expiresAt: z.string().datetime(),
    })

    const UpdateJobSchema = z.object({
      title: z.string().min(1).max(200).optional(),
      description: z.string().min(1).max(5000).optional(),
      professionId: z.string().min(1).optional(),
      salaryMin: z.number().positive().optional(),
      salaryMax: z.number().positive().optional(),
      lat: z.number().min(-90).max(90).optional(),
      lng: z.number().min(-180).max(180).optional(),
      modality: z.enum(['REMOTE', 'HYBRID', 'IN_PERSON']).optional(),
      requiredSkills: z.array(z.string()).min(1).optional(),
      expiresAt: z.string().datetime().optional(),
    })

    export async function jobRoutes(app: FastifyInstance): Promise<void> {
      const repo = new JobRepository()
      const service = new JobService(repo)

      // POST /jobs — create a new job posting
      app.post('/', {
        preHandler: [authenticate, requireRole('RECRUITER')],
      }, async (request, reply) => {
        const body = CreateJobSchema.parse(request.body)
        const job = await service.createJob(request.user.id, {
          ...body,
          expiresAt: new Date(body.expiresAt),
        })
        return reply.status(201).send(job)
      })

      // GET /jobs/mine — list recruiter's own jobs (MUST be before /:id)
      app.get('/mine', {
        preHandler: [authenticate, requireRole('RECRUITER')],
      }, async (request, reply) => {
        const jobs = await service.listJobsByRecruiter(request.user.id)
        return reply.status(200).send(jobs)
      })

      // GET /jobs/:id — job detail with placeholder rankings
      app.get('/:id', {
        preHandler: [authenticate, requireRole('RECRUITER')],
      }, async (request, reply) => {
        const { id } = request.params as { id: string }
        const job = await service.getJobById(id)
        // rankings: [] — placeholder; Phase 6 fills this with fit-scored candidates
        return reply.status(200).send({ ...job, rankings: [] })
      })

      // PATCH /jobs/:id — update job details
      app.patch('/:id', {
        preHandler: [authenticate, requireRole('RECRUITER')],
      }, async (request, reply) => {
        const { id } = request.params as { id: string }
        const body = UpdateJobSchema.parse(request.body)
        const job = await service.updateJob(request.user.id, id, {
          ...body,
          expiresAt: body.expiresAt ? new Date(body.expiresAt) : undefined,
        })
        return reply.status(200).send(job)
      })

      // DELETE /jobs/:id/deactivate — soft deactivation
      app.delete('/:id/deactivate', {
        preHandler: [authenticate, requireRole('RECRUITER')],
      }, async (request, reply) => {
        const { id } = request.params as { id: string }
        const job = await service.deactivateJob(request.user.id, id)
        return reply.status(200).send(job)
      })
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → POST /jobs | Untrusted: title, description, skills, coordinates, salary, companyId |
| JWT payload → service | userId from JWT is trusted after authenticate hook; role enforced by requireRole |
| Route handler → JobService | jobId from URL params is untrusted — service validates existence and ownership |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-03-01 | Spoofing | POST /jobs — companyId in body | mitigate | recruiterId is derived from JWT userId via profile lookup (service layer) — even if attacker sends a valid companyId for another company, the job is owned by their profile's company. Note: companyId in body is used as provided; Phase 4 does not cross-validate companyId against recruiter's company. This is acceptable for v1 — recruiters create jobs for their linked company |
| T-04-03-02 | Tampering (IDOR) | PATCH /jobs/:id, DELETE /jobs/:id/deactivate | mitigate | service.updateJob and service.deactivateJob compare job.recruiterId to authenticated user's profile.id — 403 thrown on mismatch (tested in integration suite) |
| T-04-03-03 | Elevation of Privilege | All /jobs routes | mitigate | requireRole('RECRUITER') preHandler blocks CANDIDATE and ADMIN — ASVS L1; covered by 401/403 integration tests |
| T-04-03-04 | Information Disclosure | GET /jobs/:id rankings field | accept | rankings: [] placeholder reveals no candidate information — Phase 6 will add access control when real data is added |
| T-04-03-05 | Injection | Zod input validation | mitigate | All body fields validated via Zod schemas before reaching service — SEC-02 compliance; coordinates bounded to valid lat/lng ranges |
| T-04-03-06 | Tampering | expiresAt string coercion | mitigate | z.string().datetime() enforces ISO8601 format; coerced to Date before Prisma — no raw date string reaches the DB |
</threat_model>
