```typescript
import { FastifyInstance } from 'fastify'
import { z } from 'zod'
import { authenticate } from '../../shared/hooks/authenticate'
import { requireRole } from '../../shared/hooks/requireRole'
import { CreditRepository } from './credits.repository'
import { CreditService } from './credits.service'
import { haversineKm } from '../../shared/utils/haversine'

const UnlockParamsSchema = z.object({
  jobId: z.string().uuid(),
  candidateId: z.string().uuid(),
})

export async function creditsJobRoutes(app: FastifyInstance): Promise<void> {
  const repo = new CreditRepository()
  const service = new CreditService(repo)

  // POST /jobs/:jobId/candidates/:candidateId/unlock
  // RECRUITER only — D-04
  app.post('/:jobId/candidates/:candidateId/unlock', {
    preHandler: [authenticate, requireRole('RECRUITER')],
  }, async (request, reply) => {
    const { jobId, candidateId } = UnlockParamsSchema.parse(request.params)

    // Service handles all guard checks (ownership, swipe direction, balance, double-unlock)
    const unlock = await service.unlockCandidate(request.user.id, jobId, candidateId)

    // Build full profile response per D-05
    // Fetch candidate profile for response (service returns Unlock record, not full profile)
    const candidateProfile = await repo.findCandidateProfileById(candidateId)
    const job = await repo.findJobById(jobId)

    const distanceKm = candidateProfile && job
      ? haversineKm(candidateProfile.lat, candidateProfile.lng, job.lat ?? null, job.lng ?? null)
      : null

    return reply.status(201).send({
      id: unlock.id,
      companyId: unlock.companyId,
      candidateId: unlock.candidateId,
      jobId: unlock.jobId,
      unlockedAt: unlock.unlockedAt,
      creditsCost: unlock.creditsCost,
      profile: candidateProfile
        ? {
            name: candidateProfile.name,
            avatarUrl: candidateProfile.user.avatarUrl,
            bio: candidateProfile.bio,
            skills: candidateProfile.skills,
            professionId: candidateProfile.professionId,
            preferredModalities: candidateProfile.preferredModalities,
            salaryExpectation: candidateProfile.salaryExpectation,
            distanceKm,
          }
        : null,
    })
  })
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → credits.routes.ts | Untrusted request; params validated with Zod, auth enforced by preHandlers |
| credits.routes.ts → CreditService | Service receives only validated primitives (userId from JWT, jobId/candidateId from Zod) |
| CreditService → prisma.$transaction | DB boundary; balance check and writes are atomic |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-01 | Tampering (TOCTOU) | CreditRepository.executeUnlock | mitigate | Balance check `company.credits < creditsCost` is INSIDE `prisma.$transaction` callback (first operation); verified by integration test 402 path |
