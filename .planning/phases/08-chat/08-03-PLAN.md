**Create src/modules/chat/chat.routes.ts from scratch:**

This is the first WebSocket route in the codebase. The WS upgrade route uses `{ websocket: true }` and `preHandler: [authenticate]`. The in-memory connection store is a module-level singleton Map. REST routes use `[authenticate, requireRole('CANDIDATE', 'RECRUITER')]`.

```typescript
import { FastifyInstance } from 'fastify'
import { z } from 'zod'
import type { WebSocket } from '@fastify/websocket'
import { authenticate } from '../../shared/hooks/authenticate'
import { requireRole } from '../../shared/hooks/requireRole'
import type { IChatService } from '../../shared/interfaces/chat.interface'

// In-memory connection store — module-level singleton (D-10)
// Map<chatId, Set<WebSocket>>
const connections = new Map<string, Set<WebSocket>>()

function addConnection(chatId: string, socket: WebSocket): void {
  if (!connections.has(chatId)) {
    connections.set(chatId, new Set())
  }
  connections.get(chatId)!.add(socket)
}

function removeConnection(chatId: string, socket: WebSocket): void {
  const room = connections.get(chatId)
  if (room) {
    room.delete(socket)
    if (room.size === 0) connections.delete(chatId) // Prevent memory leak (RESEARCH Pitfall 4)
  }
}

function broadcast(chatId: string, message: object): void {
  const room = connections.get(chatId)
  if (!room) return
  const json = JSON.stringify(message)
  for (const socket of room) {
    if (socket.readyState === socket.OPEN) {
      socket.send(json)
    }
  }
}

// Zod schemas
const ChatIdParamsSchema = z.object({ chatId: z.string().uuid() })
const MessagesQuerySchema = z.object({
  cursor: z.string().uuid().optional(),
  limit: z.coerce.number().int().positive().max(100).default(20),
})
const IncomingMessageSchema = z.object({
  content: z.string().min(1).max(5000),
})

export async function chatRoutes(
  app: FastifyInstance,
  opts: { service: IChatService & {
    verifyParticipant: (chatId: string, userId: string) => Promise<boolean>
    getMessages: (chatId: string, userId: string, cursor: string | undefined, limit: number) => Promise<{ data: object[]; nextCursor: string | null }>
    listChats: (userId: string) => Promise<object[]>
  } },
): Promise<void> {
  const service = opts.service

  // GET /chats — list chats for authenticated user (D-12)
  app.get('/', {
    preHandler: [authenticate, requireRole('CANDIDATE', 'RECRUITER')],
  }, async (request, reply) => {
    const chats = await service.listChats(request.user.id)
    return reply.status(200).send(chats)
  })

  // GET /chats/:chatId/messages — paginated message history (D-13, D-14)
  app.get('/:chatId/messages', {
    preHandler: [authenticate, requireRole('CANDIDATE', 'RECRUITER')],
  }, async (request, reply) => {
    const { chatId } = ChatIdParamsSchema.parse(request.params)
    const { cursor, limit } = MessagesQuerySchema.parse(request.query)
    const result = await service.getMessages(chatId, request.user.id, cursor, limit)
    return reply.status(200).send(result)
  })

  // GET /chats/:chatId/ws — WebSocket upgrade (D-08, D-09, D-10, D-11)
  app.get('/:chatId/ws', {
    websocket: true,
    preHandler: [authenticate], // JWT verify BEFORE upgrade; participant check in handler body
  }, async (socket: WebSocket, request) => {
    const { chatId } = ChatIdParamsSchema.parse(request.params)

    // Buffer pending messages FIRST (synchronous) — prevents dropping messages
    // that arrive while async participant check is running (RESEARCH Pitfall 2)
    const pending: Buffer[] = []
    socket.on('message', (data) => pending.push(data as Buffer))

    // Async participant check (D-09)
    const isParticipant = await service.verifyParticipant(chatId, request.user.id)
    if (!isParticipant) {
      socket.close(1008, 'Forbidden')
      return
    }

    addConnection(chatId, socket)

    // Real message handler: validate → persist → broadcast (D-11)
    const handleMessage = async (rawData: Buffer): Promise<void> => {
      let payload: { content: string }
      try {
        payload = IncomingMessageSchema.parse(JSON.parse(rawData.toString()))
      } catch {
        socket.send(JSON.stringify({ error: 'INVALID_PAYLOAD' }))
        return
      }
      const message = await service.sendMessage(chatId, request.user.id, payload.content)
      broadcast(chatId, message) // Broadcast AFTER persist (D-11)
    }

    // Swap buffer listener to real handler, drain buffered messages
    socket.removeAllListeners('message')
    socket.on('message', handleMessage)
    for (const data of pending) await handleMessage(data)

    socket.on('close', () => removeConnection(chatId, socket))
    socket.on('error', () => removeConnection(chatId, socket))
  })
}
```

import type { IChatService } from '../../shared/interfaces/chat.interface'
import { ChatInitiator } from '@prisma/client'

export class CreditService {
  constructor(
    private readonly repo: CreditRepository,
    private readonly chat: IChatService, // NEW — injected at composition root (D-01)
  ) {}

  async unlockCandidate(userId: string, jobId: string, candidateId: string) {
    const recruiter = await this.repo.findRecruiterByUserId(userId)
    if (!recruiter) throw new AppError('Forbidden', 403, 'FORBIDDEN')

    const job = await this.repo.findJobById(jobId)
    if (!job) throw new AppError('Job not found', 404, 'NOT_FOUND')
    if (job.recruiterId !== recruiter.id) throw new AppError('Forbidden', 403, 'FORBIDDEN')

    const swipe = await this.repo.findSwipe(candidateId, jobId)
    if (!swipe) throw new AppError('Candidate not found on this job', 404, 'NOT_FOUND')
    if (swipe.direction !== 'DISLIKE') throw new AppError('Candidate is already visible', 422, 'CANDIDATE_ALREADY_VISIBLE')

    const existing = await this.repo.findUnlock(recruiter.companyId, candidateId, jobId)
    if (existing) throw new AppError('Already unlocked', 409, 'ALREADY_UNLOCKED')

    const cost = UNLOCK_COST_BY_PLAN[recruiter.company.plan] ?? UNLOCK_COST_BY_PLAN['FREE']

    let unlock: Awaited<ReturnType<typeof this.repo.executeUnlock>>
    try {
      unlock = await this.repo.executeUnlock(recruiter.companyId, candidateId, jobId, cost)
    } catch (err) {
      if (err instanceof Prisma.PrismaClientKnownRequestError && err.code === 'P2002') {
        throw new AppError('Already unlocked', 409, 'ALREADY_UNLOCKED')
      }
      if (err instanceof Error && err.message === 'INSUFFICIENT_CREDITS') {
        throw new AppError('Insufficient credits', 402, 'INSUFFICIENT_CREDITS')
      }
      if (err instanceof Error && err.message === 'COMPANY_NOT_FOUND') {
        throw new AppError('Company not found', 404, 'NOT_FOUND')
      }
      throw err
    }

    const { chatId } = await this.chat.createOrGetChat(
      candidateId,
      recruiter.companyId,
      jobId,
      ChatInitiator.CREDIT,
    )

    return { ...unlock, chatId } // D-03: chatId included in unlock response
  }

  async grantCredits(companyId: string, amount: number, reason = 'admin_grant') {
    if (!Number.isInteger(amount) || amount <= 0) {
      throw new AppError('Amount must be a positive integer', 400, 'INVALID_AMOUNT')
    }
    return this.repo.grantCredits(companyId, amount, reason)
  }
}
