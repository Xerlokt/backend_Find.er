**2. src/adapters/services/jwt.adapter.ts:**

```typescript
import { FastifyInstance } from 'fastify'
import { randomBytes, createHash } from 'crypto'
import { ITokenPort } from '../../application/ports/token.port'

export class JwtAdapter implements ITokenPort {
  constructor(private readonly app: FastifyInstance) {}

  signAccess(payload: { id: string; role: string }): string {
    return this.app.jwt.sign(payload, { expiresIn: '15m' })
  }

  generateRefreshToken(): { raw: string; hashed: string } {
    const raw = randomBytes(32).toString('hex')
    const hashed = createHash('sha256').update(raw).digest('hex')
    return { raw, hashed }
  }

  verifyAccess(token: string): { id: string; role: string } {
    return this.app.jwt.verify(token)
  }
}
```

NOTES:
- Constructor takes `FastifyInstance` — the @fastify/jwt plugin decorates the FastifyInstance; you cannot call `app.jwt.sign` without a real FastifyInstance
- `expiresIn: '15m'` — enforced by CLAUDE.md (JWT access token <= 15 min)
- `generateRefreshToken` returns `{ raw, hashed }` matching ITokenPort (NOT `{ rawToken, tokenHash }` which was the old auth.service.ts convention)
- `JwtAdapter` is instantiated inside each controller handler as `new JwtAdapter(request.server)` — `request.server` is the FastifyInstance on every FastifyRequest
- `crypto` is a Node.js built-in — no import needed from npm; import directly from `'crypto'`

**1. src/adapters/repositories/user.repository.ts:**

```typescript
import { Role } from '@prisma/client'
import { prisma } from '../../infrastructure/database/prisma'
import { IUserRepository } from '../../domain/interfaces/user.repository.interface'
import { User, UserRole } from '../../domain/entities/user.entity'

export class UserRepository implements IUserRepository {
  async findByEmail(email: string): Promise<User | null> {
    const row = await prisma.user.findUnique({ where: { email } })
    return row ? this.toDomain(row) : null
  }

  async findById(id: string): Promise<User | null> {
    const row = await prisma.user.findUnique({ where: { id } })
    return row ? this.toDomain(row) : null
  }

  async create(data: { email: string; passwordHash: string; role: UserRole }): Promise<User> {
    const row = await prisma.user.create({ data: { ...data, role: data.role as Role } })
    return this.toDomain(row)
  }

  async updateGoogleId(id: string, googleId: string, avatarUrl?: string): Promise<User> {
    const row = await prisma.user.update({
      where: { id },
      data: { googleId, ...(avatarUrl !== undefined ? { avatarUrl } : {}) },
    })
    return this.toDomain(row)
  }

  async findOrCreateGoogle(data: {
    googleId: string
    email: string
    avatarUrl?: string
    role: UserRole
  }): Promise<{ user: User; isNew: boolean }> {
    return prisma.$transaction(async (tx) => {
      // Try to find by googleId first (already linked)
      const byGoogleId = await tx.user.findUnique({ where: { googleId: data.googleId } })
      if (byGoogleId) return { user: this.toDomain(byGoogleId), isNew: false }

      // Link googleId to existing account by email
      const existing = await tx.user.findUnique({ where: { email: data.email } })
      if (existing) {
        const updated = await tx.user.update({
          where: { id: existing.id },
          data: {
            googleId: data.googleId,
            avatarUrl: data.avatarUrl ?? existing.avatarUrl,
          },
        })
        return { user: this.toDomain(updated), isNew: false }
      }

      // First Google auth — create new user
      const created = await tx.user.create({
        data: {
          email: data.email,
          googleId: data.googleId,
          avatarUrl: data.avatarUrl,
          role: data.role as Role,
        },
      })
      return { user: this.toDomain(created), isNew: true }
    })
  }

  private toDomain(row: {
    id: string
    email: string
    passwordHash: string | null
    googleId: string | null
    avatarUrl: string | null
    role: Role
    createdAt: Date
  }): User {
    return {
      id: row.id,
      email: row.email,
      passwordHash: row.passwordHash,
      googleId: row.googleId,
      avatarUrl: row.avatarUrl,
      role: row.role as UserRole,
      createdAt: row.createdAt,
    }
  }
}
```

NOTES:
- Import `prisma` from `'../../infrastructure/database/prisma'` — NOT from `'../../shared/db/prisma'`
- `Role` from `@prisma/client` is needed for the Prisma role cast — this is the ONLY acceptable @prisma/client import in adapter files (it is a cast, not a DB query type)
- `toDomain` maps ALL 7 fields explicitly — do NOT spread Prisma model objects
- `findOrCreateGoogle` drops the `name` field compared to `AuthRepository.findOrCreateGoogleUser` because `IUserRepository.findOrCreateGoogle` does not include it
- `updateGoogleId` only updates avatarUrl conditionally (when provided) — matches the interface signature `avatarUrl?: string`

**2. src/adapters/repositories/refresh-token.repository.ts:**

```typescript
import { prisma } from '../../infrastructure/database/prisma'
import { IRefreshTokenRepository } from '../../domain/interfaces/refresh-token.repository.interface'
import { RefreshToken } from '../../domain/entities/refresh-token.entity'

export class RefreshTokenRepository implements IRefreshTokenRepository {
  async create(data: {
    tokenHash: string
    familyId: string
    userId: string
    expiresAt: Date
  }): Promise<RefreshToken> {
    const row = await prisma.refreshToken.create({ data })
    return this.toDomain(row)
  }

  async findByHash(tokenHash: string): Promise<RefreshToken | null> {
    const row = await prisma.refreshToken.findUnique({ where: { tokenHash } })
    return row ? this.toDomain(row) : null
  }

  async revoke(id: string): Promise<void> {
    await prisma.refreshToken.update({
      where: { id },
      data: { revokedAt: new Date() },
    })
  }

  async revokeFamily(familyId: string): Promise<void> {
    await prisma.refreshToken.updateMany({
      where: { familyId, revokedAt: null },
      data: { revokedAt: new Date() },
    })
  }

  async revokeByHash(tokenHash: string): Promise<number> {
    const result = await prisma.refreshToken.updateMany({
      where: { tokenHash, revokedAt: null },
      data: { revokedAt: new Date() },
    })
    return result.count
  }

  private toDomain(row: {
    id: string
    tokenHash: string
    familyId: string
    userId: string
    expiresAt: Date
    revokedAt: Date | null
    createdAt: Date
  }): RefreshToken {
    return {
      id: row.id,
      tokenHash: row.tokenHash,
      familyId: row.familyId,
      userId: row.userId,
      expiresAt: row.expiresAt,
      revokedAt: row.revokedAt,
      createdAt: row.createdAt,
    }
  }
}
```

NOTES:
- Import `prisma` from `'../../infrastructure/database/prisma'` — NOT from `'../../shared/db/prisma'`
- `revokeByHash` returns `result.count` — this is the atomic replay-attack detection number. The `updateMany` is a single SQL UPDATE; its `count` tells callers whether they were the one to revoke (1) or it was already revoked (0 = replay)
- `toDomain` maps ALL 7 fields explicitly
- No `@prisma/client` import needed — refreshToken model fields match domain entity types directly
