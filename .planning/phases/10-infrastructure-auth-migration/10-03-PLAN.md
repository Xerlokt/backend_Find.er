```typescript
import { randomBytes } from 'crypto'
import { IUserRepository } from '../../../domain/interfaces/user.repository.interface'
import { IRefreshTokenRepository } from '../../../domain/interfaces/refresh-token.repository.interface'
import { IHashPort } from '../../ports/hash.port'
import { ITokenPort } from '../../ports/token.port'
import { DomainError } from '../../errors/domain.error'

interface RegisterInput {
  email: string
  password: string
  role: 'CANDIDATE' | 'RECRUITER'
}

export class RegisterUseCase {
  constructor(
    private readonly userRepo: IUserRepository,
    private readonly refreshTokenRepo: IRefreshTokenRepository,
    private readonly hash: IHashPort,
    private readonly token: ITokenPort,
  ) {}

  async execute(input: RegisterInput): Promise<{ accessToken: string; refreshToken: string }> {
    const existing = await this.userRepo.findByEmail(input.email)
    if (existing) {
      throw new DomainError('Email already in use', 'CONFLICT', 409)
    }

    const passwordHash = await this.hash.hash(input.password)
    const user = await this.userRepo.create({ email: input.email, passwordHash, role: input.role })

    const accessToken = this.token.signAccess({ id: user.id, role: user.role })
    const { raw, hashed } = this.token.generateRefreshToken()
    const familyId = randomBytes(16).toString('hex')
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    await this.refreshTokenRepo.create({ tokenHash: hashed, familyId, userId: user.id, expiresAt })

    return { accessToken, refreshToken: raw }
  }
}
```

```typescript
import { randomBytes } from 'crypto'
import { IUserRepository } from '../../../domain/interfaces/user.repository.interface'
import { IRefreshTokenRepository } from '../../../domain/interfaces/refresh-token.repository.interface'
import { IHashPort } from '../../ports/hash.port'
import { ITokenPort } from '../../ports/token.port'
import { DomainError } from '../../errors/domain.error'

// Generated via: bcryptjs.hashSync('__dummy__', 12)
// Timing safety: bcrypt.compare always runs a full comparison even when user not found
const DUMMY_HASH = '$2b$12$LVfxFJGoT4lbSwHhLU8kCOy8xQpBLaIH5TNXLilzj5Wky3l2KBKP2'

interface LoginInput {
  email: string
  password: string
}

export class LoginUseCase {
  constructor(
    private readonly userRepo: IUserRepository,
    private readonly refreshTokenRepo: IRefreshTokenRepository,
    private readonly hash: IHashPort,
    private readonly token: ITokenPort,
  ) {}

  async execute(input: LoginInput): Promise<{ accessToken: string; refreshToken: string }> {
    const user = await this.userRepo.findByEmail(input.email)

    // Timing-safe: always run compare even when user not found (prevents timing oracle attack)
    const passwordMatch = await this.hash.compare(
      input.password,
      user?.passwordHash ?? DUMMY_HASH,
    )

    if (!user || !passwordMatch) {
      throw new DomainError('Invalid credentials', 'INVALID_CREDENTIALS', 401)
    }

    const accessToken = this.token.signAccess({ id: user.id, role: user.role })
    const { raw, hashed } = this.token.generateRefreshToken()
    const familyId = randomBytes(16).toString('hex')
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    await this.refreshTokenRepo.create({ tokenHash: hashed, familyId, userId: user.id, expiresAt })

    return { accessToken, refreshToken: raw }
  }
}
```

```typescript
import { createHash } from 'crypto'
import { IRefreshTokenRepository } from '../../../domain/interfaces/refresh-token.repository.interface'

export class LogoutUseCase {
  constructor(
    private readonly refreshTokenRepo: IRefreshTokenRepository,
  ) {}

  async execute(rawToken: string): Promise<void> {
    const tokenHash = createHash('sha256').update(rawToken).digest('hex')
    const stored = await this.refreshTokenRepo.findByHash(tokenHash)

    if (!stored || stored.revokedAt !== null) {
      // Idempotent — token not found or already revoked; no error thrown
      return
    }

    // Revoke entire family — invalidates all active sessions
    await this.refreshTokenRepo.revokeFamily(stored.familyId)
  }
}
```

**1. src/application/use-cases/auth/refresh-token.use-case.ts:**

```typescript
import { createHash } from 'crypto'
import { IUserRepository } from '../../../domain/interfaces/user.repository.interface'
import { IRefreshTokenRepository } from '../../../domain/interfaces/refresh-token.repository.interface'
import { ITokenPort } from '../../ports/token.port'
import { DomainError } from '../../errors/domain.error'

export class RefreshTokenUseCase {
  constructor(
    private readonly userRepo: IUserRepository,
    private readonly refreshTokenRepo: IRefreshTokenRepository,
    private readonly token: ITokenPort,
  ) {}

  async execute(rawToken: string): Promise<{ accessToken: string; refreshToken: string }> {
    const tokenHash = createHash('sha256').update(rawToken).digest('hex')

    // Atomic revoke — Prisma updateMany is a single SQL UPDATE.
    // count=1: we revoked it (valid use). count=0: already revoked (replay attack).
    const count = await this.refreshTokenRepo.revokeByHash(tokenHash)
    const stored = await this.refreshTokenRepo.findByHash(tokenHash)

    if (!stored) {
      throw new DomainError('Invalid token', 'UNAUTHORIZED', 401)
    }

    if (count === 0) {
      // Replay attack detected — revoke entire token family to invalidate all sessions
      await this.refreshTokenRepo.revokeFamily(stored.familyId)
      throw new DomainError('Token revoked', 'TOKEN_EXPIRED', 401)
    }

    if (stored.expiresAt < new Date()) {
      throw new DomainError('Token expired', 'TOKEN_EXPIRED', 401)
    }

    const user = await this.userRepo.findById(stored.userId)
    if (!user) {
      throw new DomainError('User not found', 'UNAUTHORIZED', 401)
    }

    const accessToken = this.token.signAccess({ id: user.id, role: user.role })
    const { raw, hashed } = this.token.generateRefreshToken()
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    await this.refreshTokenRepo.create({
      tokenHash: hashed,
      familyId: stored.familyId,
      userId: user.id,
      expiresAt,
    })

    return { accessToken, refreshToken: raw }
  }
}
```

CRITICAL NOTES for refresh-token.use-case.ts:
- `revokeByHash` is called BEFORE `findByHash` — this is the atomic pattern. `updateMany` is a single SQL UPDATE; its count reveals whether this request was the legitimate revoker.
- The two calls (revokeByHash + findByHash) are NOT wrapped in a transaction in the use case. The atomicity comes from `revokeByHash`'s `updateMany` itself — two concurrent requests both do `updateMany`, only one gets count=1, the loser gets count=0. `findByHash` after is for reading metadata (familyId, expiresAt, userId), not for atomicity.
- The new refresh token uses `stored.familyId` (continuation of the token family), NOT a new familyId.
- Zero imports from `@prisma/client` — the `prisma.$transaction` from the old auth.service.ts is replaced by the revokeByHash count pattern.

```typescript
import { randomBytes } from 'crypto'
import { OAuth2Client } from 'google-auth-library'
import { IUserRepository } from '../../../domain/interfaces/user.repository.interface'
import { IRefreshTokenRepository } from '../../../domain/interfaces/refresh-token.repository.interface'
import { ITokenPort } from '../../ports/token.port'
import { DomainError } from '../../errors/domain.error'

// OAuth2Client stays in use case for Phase 10 — no IGoogleAuthPort port yet (deferred)
const googleClient = new OAuth2Client(process.env.GOOGLE_CLIENT_ID)

interface LoginWithGoogleInput {
  idToken: string
  role: 'CANDIDATE' | 'RECRUITER'
}

export class LoginWithGoogleUseCase {
  constructor(
    private readonly userRepo: IUserRepository,
    private readonly refreshTokenRepo: IRefreshTokenRepository,
    private readonly token: ITokenPort,
  ) {}

  async execute(input: LoginWithGoogleInput): Promise<{ accessToken: string; refreshToken: string }> {
    let googlePayload: { sub: string; email: string; picture?: string }

    try {
      const ticket = await googleClient.verifyIdToken({
        idToken: input.idToken,
        audience: process.env.GOOGLE_CLIENT_ID,
      })
      const payload = ticket.getPayload()!
      googlePayload = {
        sub: payload.sub,
        email: payload.email!,
        picture: payload.picture,
      }
    } catch {
      throw new DomainError('Invalid Google token', 'INVALID_CREDENTIALS', 401)
    }

    const { user } = await this.userRepo.findOrCreateGoogle({
      googleId: googlePayload.sub,
      email: googlePayload.email,
      avatarUrl: googlePayload.picture,
      role: input.role,
    })

    const accessToken = this.token.signAccess({ id: user.id, role: user.role })
    const { raw, hashed } = this.token.generateRefreshToken()
    const familyId = randomBytes(16).toString('hex')
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    await this.refreshTokenRepo.create({ tokenHash: hashed, familyId, userId: user.id, expiresAt })

    return { accessToken, refreshToken: raw }
  }
}
```
