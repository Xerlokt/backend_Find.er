**1. src/adapters/controllers/auth.controller.ts** (full file):

```typescript
import { FastifyRequest, FastifyReply } from 'fastify'
import { z } from 'zod'
import { RegisterUseCase } from '../../application/use-cases/auth/register.use-case'
import { LoginUseCase } from '../../application/use-cases/auth/login.use-case'
import { RefreshTokenUseCase } from '../../application/use-cases/auth/refresh-token.use-case'
import { LogoutUseCase } from '../../application/use-cases/auth/logout.use-case'
import { LoginWithGoogleUseCase } from '../../application/use-cases/auth/login-with-google.use-case'
import { UserRepository } from '../repositories/user.repository'
import { RefreshTokenRepository } from '../repositories/refresh-token.repository'
import { BcryptAdapter } from '../services/bcrypt.adapter'
import { JwtAdapter } from '../services/jwt.adapter'
import { DomainError } from '../../application/errors/domain.error'

const RegisterBodySchema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
  role: z.enum(['CANDIDATE', 'RECRUITER']),
})

const LoginBodySchema = z.object({
  email: z.string().email(),
  password: z.string().min(1),
})

const RefreshBodySchema = z.object({
  refreshToken: z.string().min(1),
})

const LogoutBodySchema = z.object({
  refreshToken: z.string().min(1),
})

const GoogleBodySchema = z.object({
  idToken: z.string().min(1),
  role: z.enum(['CANDIDATE', 'RECRUITER']),
})

export class AuthController {
  async register(request: FastifyRequest, reply: FastifyReply): Promise<void> {
    const body = RegisterBodySchema.parse(request.body)
    const useCase = new RegisterUseCase(
      new UserRepository(),
      new RefreshTokenRepository(),
      new BcryptAdapter(),
      new JwtAdapter(request.server),
    )
    try {
      const tokens = await useCase.execute(body)
      reply.status(201).send(tokens)
    } catch (err) {
      if (err instanceof DomainError) {
        reply.status(err.statusCode).send({ error: err.message, code: err.code })
        return
      }
      throw err
    }
  }

  async login(request: FastifyRequest, reply: FastifyReply): Promise<void> {
    const body = LoginBodySchema.parse(request.body)
    const useCase = new LoginUseCase(
      new UserRepository(),
      new RefreshTokenRepository(),
      new BcryptAdapter(),
      new JwtAdapter(request.server),
    )
    try {
      const tokens = await useCase.execute(body)
      reply.status(200).send(tokens)
    } catch (err) {
      if (err instanceof DomainError) {
        reply.status(err.statusCode).send({ error: err.message, code: err.code })
        return
      }
      throw err
    }
  }

  async refresh(request: FastifyRequest, reply: FastifyReply): Promise<void> {
    const body = RefreshBodySchema.parse(request.body)
    const useCase = new RefreshTokenUseCase(
      new UserRepository(),
      new RefreshTokenRepository(),
      new JwtAdapter(request.server),
    )
    try {
      const tokens = await useCase.execute(body.refreshToken)
      reply.status(200).send(tokens)
    } catch (err) {
      if (err instanceof DomainError) {
        reply.status(err.statusCode).send({ error: err.message, code: err.code })
        return
      }
      throw err
    }
  }

  async logout(request: FastifyRequest, reply: FastifyReply): Promise<void> {
    const body = LogoutBodySchema.parse(request.body)
    const useCase = new LogoutUseCase(
      new RefreshTokenRepository(),
    )
    try {
      await useCase.execute(body.refreshToken)
      reply.status(204).send()
    } catch (err) {
      if (err instanceof DomainError) {
        reply.status(err.statusCode).send({ error: err.message, code: err.code })
        return
      }
      throw err
    }
  }

  async loginWithGoogle(request: FastifyRequest, reply: FastifyReply): Promise<void> {
    const body = GoogleBodySchema.parse(request.body)
    const useCase = new LoginWithGoogleUseCase(
      new UserRepository(),
      new RefreshTokenRepository(),
      new JwtAdapter(request.server),
    )
    try {
      const tokens = await useCase.execute(body)
      reply.status(201).send(tokens)
    } catch (err) {
      if (err instanceof DomainError) {
        reply.status(err.statusCode).send({ error: err.message, code: err.code })
        return
      }
      throw err
    }
  }
}
```

NOTES for auth.controller.ts:
- `JwtAdapter(request.server)` — `request.server` is the FastifyInstance on every FastifyRequest; this is how @fastify/jwt decorations are accessible
- Each method has its own Zod parse + use case instantiation — no shared state on the controller class (stateless)
- The DomainError catch is in EVERY handler — even though errorHandler.ts now handles DomainError too, the explicit catch in the controller is a defense-in-depth pattern. It maps DomainError before Fastify's errorHandler sees it. Both paths produce identical wire responses.
- `logout` uses `LogoutBodySchema` (same shape as RefreshBodySchema — refreshToken field) — parse explicitly to avoid `request.body as any`
- The controller contains NO business logic — it only: parses, instantiates, calls, maps errors, sends

**2. src/infrastructure/http/routes/auth.routes.ts** (full file):

Create the `src/infrastructure/http/routes/` directory and the file:

```typescript
import { FastifyInstance } from 'fastify'
import { AuthController } from '../../../adapters/controllers/auth.controller'

export async function authRoutes(app: FastifyInstance): Promise<void> {
  const ctrl = new AuthController()

  app.post('/register', {
    config: { rateLimit: { max: 3, timeWindow: '1 minute' } },
  }, ctrl.register.bind(ctrl))

  app.post('/login', {
    config: { rateLimit: { max: 5, timeWindow: '1 minute' } },
  }, ctrl.login.bind(ctrl))

  app.post('/refresh', {
    config: { rateLimit: { max: 10, timeWindow: '1 minute' } },
  }, ctrl.refresh.bind(ctrl))

  app.post('/logout', ctrl.logout.bind(ctrl))

  app.post('/google', {
    config: { rateLimit: { max: 5, timeWindow: '1 minute' } },
  }, ctrl.loginWithGoogle.bind(ctrl))
}
```

NOTES for auth.routes.ts:
- ZERO Zod, ZERO business logic, ZERO service instantiation — only path + rate limit config + delegate
- Export function name is `authRoutes` — must match what src/app.ts imports
- `.bind(ctrl)` is required because Fastify calls the handler without the class context; without bind, `this` inside the handler would be undefined
- Rate limits match the existing `src/modules/auth/auth.routes.ts` exactly (3/5/10/5 req/min)
