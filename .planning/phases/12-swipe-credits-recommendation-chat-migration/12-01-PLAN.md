```typescript
import { IChatRepository } from '../../../domain/interfaces/chat.repository.interface'
import { ChatInitiator } from '../../../domain/entities/chat.entity'
export class CreateOrGetChatUseCase {
  constructor(private readonly repo: IChatRepository) {}
  async execute(candidateId: string, companyId: string, jobId: string, initiator: ChatInitiator): Promise<{ chatId: string }> {
    const chat = await this.repo.createOrGetChat(candidateId, companyId, jobId, initiator)
    return { chatId: chat.id }
  }
}
```

- Receives IChatRepository
- execute(chatId, senderId, content): findParticipants → null → DomainError('Chat not found', 'NOT_FOUND', 404); isParticipant check → DomainError('Not a participant', 'NOT_PARTICIPANT', 403); countRecruiterMessages → 0 && sender !== recruiterUserId → DomainError('Recruiter must send first message', 'RECRUITER_MUST_INITIATE', 403); → repo.createMessage(chatId, senderId, content)

verify-participant.use-case.ts:
- execute(chatId, userId): findParticipants → null → return false; return userId === candidateUserId || userId === recruiterUserId

get-messages.use-case.ts:

list-chats.use-case.ts:
