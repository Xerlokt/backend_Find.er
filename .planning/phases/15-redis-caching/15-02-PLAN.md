import neo4j, { Driver } from 'neo4j-driver'

let driver: Driver

export function getNeo4jDriver(): Driver {
  if (!driver) {
    const uri = process.env.NEO4J_URI
    const user = process.env.NEO4J_USER
    const password = process.env.NEO4J_PASSWORD
    if (!uri || !user || !password) {
      throw new Error('NEO4J_URI, NEO4J_USER, and NEO4J_PASSWORD must all be set')
    }
    driver = neo4j.driver(uri, neo4j.auth.basic(user, password), { disableLosslessIntegers: true })
  }
  return driver
}

export async function closeNeo4j(): Promise<void> {
  if (driver) {
    await driver.close()
  }
}

Step 2 — Create src/adapters/services/redis.adapter.ts. Import Redis from 'ioredis' (for the constructor type only). Import ICachePort from '../../application/ports/cache.port'. Implement all 4 interface methods. Use scanStream + pipeline for invalidatePattern — NEVER use KEYS *.

```typescript
import Redis from 'ioredis'
import { ICachePort } from '../../application/ports/cache.port'

export class RedisAdapter implements ICachePort {
  constructor(private readonly client: Redis) {}

  async get(key: string): Promise<string | null> {
    return this.client.get(key)
  }

  async set(key: string, value: string, ttlSeconds: number): Promise<void> {
    await this.client.set(key, value, 'EX', ttlSeconds)
  }

  async del(key: string): Promise<void> {
    await this.client.del(key)
  }

  async invalidatePattern(pattern: string): Promise<void> {
    // SCAN + pipeline — never KEYS * (KEYS blocks the Redis event loop — see RESEARCH Pitfall 2)
    const stream = this.client.scanStream({ match: pattern, count: 100 })
    const pipeline = this.client.pipeline()
    await new Promise<void>((resolve, reject) => {
      stream.on('data', (keys: string[]) => {
        keys.forEach((key) => pipeline.del(key))
      })
      stream.on('end', () => resolve())
      stream.on('error', reject)
    })
    await pipeline.exec()
  }
}
```

Note: `check-deps.ts` allows adapters/ to import from infrastructure/database/ (the rule only forbids infrastructure/http/ — see STATE.md decision 13-03). RedisAdapter imports only from ioredis and application/ports/ — both are valid.
