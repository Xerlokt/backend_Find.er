```typescript
import { ISwipeRepository } from '../../../domain/interfaces/swipe.repository.interface'
import { IChatPort } from '../../ports/chat.port'
import { ICachePort } from '../../ports/cache.port'
import { Swipe, SwipeDirection } from '../../../domain/entities/swipe.entity'
import { DomainError } from '../../errors/domain.error'

export class RecordSwipeUseCase {
  constructor(
    private readonly repo: ISwipeRepository,
    private readonly chat: IChatPort,
    private readonly cache: ICachePort,
  ) {}

  async execute(userId: string, jobId: string, direction: SwipeDirection): Promise<Swipe & { chatId: string | null }> {
    const candidate = await this.repo.findCandidateProfileByUserId(userId)
    if (!candidate) throw new DomainError('Candidate profile not found', 'NOT_FOUND', 404)

    const job = await this.repo.findJobById(jobId)
    if (!job) throw new DomainError('Job not found', 'NOT_FOUND', 404)
    if (!job.isActive || job.expiresAt <= new Date())
      throw new DomainError('Job not found', 'NOT_FOUND', 404)

    const swipe = await this.repo.createSwipe({ candidateId: candidate.id, jobId, direction })

    try { await this.cache.del(`feed:${candidate.id}`) } catch { /* Redis down — ok */ }

    let chatId: string | null = null
    if (direction === 'LIKE') {
      const result = await this.chat.createOrGetChat(candidate.id, job.companyId, jobId, 'MATCH')
      chatId = result.chatId
    }

    return { ...swipe, chatId }
  }
}
```

```typescript
import { prisma } from '../../infrastructure/database/prisma'
import { IDiscRepository } from '../../domain/interfaces/disc.repository.interface'
import { DiscResult } from '../../domain/entities/disc-result.entity'

export class DiscRepository implements IDiscRepository {
  async upsert(
    candidateId: string,
    data: { scoreD: number; scoreI: number; scoreS: number; scoreC: number; dominant: string },
  ): Promise<DiscResult> {
    const row = await prisma.discResult.upsert({
      where: { candidateId },
      create: { candidateId, ...data },
      update: { ...data },
    })
    return this.toDomain(row)
  }

  async findByCandidateId(candidateId: string): Promise<DiscResult | null> {
    const row = await prisma.discResult.findUnique({ where: { candidateId } })
    return row ? this.toDomain(row) : null
  }

  private toDomain(row: any): DiscResult {
    return {
      id: row.id, candidateId: row.candidateId,
      scoreD: row.scoreD, scoreI: row.scoreI, scoreS: row.scoreS, scoreC: row.scoreC,
      dominant: row.dominant, createdAt: row.createdAt, updatedAt: row.updatedAt,
    }
  }
}
```

```typescript
  private static computeFactorScores(
    candidate: { professionId: string; skills: string[]; preferredModalities: string[]; salaryExpectation: number | null; lat: number | null; lng: number | null; searchRadiusKm: number },
    job: { professionId: string; requiredSkills: string[]; modality: string; salaryMin: number | null; salaryMax: number | null; lat: number | null; lng: number | null },
  ): { profession: number; skills: number; location: number; modality: number; salary: number } {
    // Factor 1: Profession (exact match)
    const profession = candidate.professionId === job.professionId ? 1.0 : 0.0

    // Factor 2: Skills (direct intersection — no Neo4j per Pitfall 2)
    const skills =
      job.requiredSkills.length === 0
        ? 1.0
        : candidate.skills.filter((s) => job.requiredSkills.includes(s)).length / job.requiredSkills.length

    // Factor 3: Location proximity
    let location: number
    if (job.lat == null || job.lng == null) {
      location = 1.0
    } else if (candidate.lat == null || candidate.lng == null) {
      location = 0.5
    } else {
      const radius = candidate.searchRadiusKm > 0 ? candidate.searchRadiusKm : 50
      const distKm = haversineKm(candidate.lat, candidate.lng, job.lat, job.lng) ?? 0
      location = Math.max(0, 1 - distKm / radius)
    }

    // Factor 4: Work mode
    const modality = candidate.preferredModalities.includes(job.modality) ? 1.0 : 0.0

    // Factor 5: Salary compatibility
    let salary: number
    if (candidate.salaryExpectation == null) {
      salary = 1.0
    } else if (job.salaryMin == null && job.salaryMax == null) {
      salary = 1.0
    } else {
      const aboveMin = job.salaryMin == null || candidate.salaryExpectation >= job.salaryMin
      const belowMax = job.salaryMax == null || candidate.salaryExpectation <= job.salaryMax
      salary = aboveMin && belowMax ? 1.0 : 0.0
    }

    return { profession, skills, location, modality, salary }
  }
```
