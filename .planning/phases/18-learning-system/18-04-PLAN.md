Follow the pattern from tests/integration/disc.test.ts. The test file uses the global setup.ts (imported via jest config), which runs beforeAll (migrate deploy) and afterEach (full cleanup).

```typescript
import request from 'supertest'
import { prisma } from '../../src/infrastructure/database/prisma'
import { buildApp } from '../../src/app'
import { FastifyInstance } from 'fastify'

let app: FastifyInstance

beforeAll(async () => {
  app = await buildApp()
  await app.ready()
})

afterAll(async () => {
  await app.close()
})

// Helper: register candidate user, create profile, return { token, candidateId }
async function setupCandidate() {
  const registerRes = await request(app.server)
    .post('/auth/register')
    .send({ email: `cand-${Date.now()}@test.com`, password: 'Password123!', role: 'CANDIDATE' })
  expect(registerRes.status).toBe(201)
  const { accessToken, userId } = registerRes.body

  const profileRes = await request(app.server)
    .post('/candidates')
    .set('Authorization', `Bearer ${accessToken}`)
    .send({
      name: 'Test Candidate',
      professionId: 'backend-dev',
      skills: ['nodejs', 'typescript'],
      preferredModalities: ['REMOTE'],
      searchRadiusKm: 50,
    })
  expect(profileRes.status).toBe(201)
  const candidateId = profileRes.body.id

  return { token: accessToken, userId, candidateId }
}

// Helper: register recruiter + company + job, return { jobId }
async function setupJob(professionId = 'backend-dev', requiredSkills = ['nodejs']) {
  const companyRes = await request(app.server)
    .post('/companies')
    .send({ name: `Company-${Date.now()}`, plan: 'BASIC' })
  const companyId = companyRes.body.id

  const recRes = await request(app.server)
    .post('/auth/register')
    .send({ email: `rec-${Date.now()}@test.com`, password: 'Password123!', role: 'RECRUITER' })
  const recToken = recRes.body.accessToken

  await request(app.server)
    .post('/recruiters')
    .set('Authorization', `Bearer ${recToken}`)
    .send({ name: 'Recruiter', companyId })

  const expiresAt = new Date(Date.now() + 86400000 * 30).toISOString()
  const jobRes = await request(app.server)
    .post('/jobs')
    .set('Authorization', `Bearer ${recToken}`)
    .send({
      title: 'Dev Job',
      description: 'Build stuff',
      professionId,
      requiredSkills,
      modality: 'REMOTE',
      expiresAt,
    })
  expect(jobRes.status).toBe(201)
  return { jobId: jobRes.body.id }
}

describe('Learning System — CandidateWeights (LEARN-01/02/03)', () => {

  it('LIKE action immediately writes CandidateWeights record with likeCount=1 (LEARN-01)', async () => {
    const { token, candidateId } = await setupCandidate()
    const { jobId } = await setupJob()

    const swipeRes = await request(app.server)
      .post('/swipes')
      .set('Authorization', `Bearer ${token}`)
      .send({ jobId, direction: 'LIKE' })
    expect(swipeRes.status).toBe(201)

    const record = await prisma.candidateWeights.findUnique({ where: { candidateId } })
    expect(record).not.toBeNull()
    expect(record!.likeCount).toBe(1)
    // Weights should have drifted from global defaults (profession match → professionWeight > 0.30 or same)
    expect(record!.professionWeight + record!.skillsWeight + record!.locationWeight +
           record!.modalityWeight + record!.salaryWeight).toBeCloseTo(1.0, 4)
  })

  it('LIKE increments likeCount on each subsequent LIKE (cold-start accumulation, LEARN-02)', async () => {
    const { token, candidateId } = await setupCandidate()

    // Create 4 different jobs and LIKE each one
    for (let i = 0; i < 4; i++) {
      const { jobId } = await setupJob()
      const res = await request(app.server)
        .post('/swipes')
        .set('Authorization', `Bearer ${token}`)
        .send({ jobId, direction: 'LIKE' })
      expect(res.status).toBe(201)
    }

    const record = await prisma.candidateWeights.findUnique({ where: { candidateId } })
    expect(record).not.toBeNull()
    expect(record!.likeCount).toBe(4)
  })

  it('feed request after 5 LIKEs uses personal weights — CandidateWeights.likeCount >= 5 (LEARN-02+03)', async () => {
    const { token, candidateId } = await setupCandidate()

    // 5 LIKEs on backend-dev jobs
    for (let i = 0; i < 5; i++) {
      const { jobId } = await setupJob('backend-dev', ['nodejs'])
      const res = await request(app.server)
        .post('/swipes')
        .set('Authorization', `Bearer ${token}`)
        .send({ jobId, direction: 'LIKE' })
      expect(res.status).toBe(201)
    }

    // Verify weights record exists with likeCount = 5
    const record = await prisma.candidateWeights.findUnique({ where: { candidateId } })
    expect(record).not.toBeNull()
    expect(record!.likeCount).toBe(5)

    // GET /jobs/feed — should succeed (personal weights used internally)
    // Create a fresh job to appear in the feed (not yet swiped)
    // Note: Neo4j may not be running in CI — so we test DB state, not live scoring
    // The key assertion is that likeCount >= 5, which triggers personal weights in GetFeedUseCase
    expect(record!.likeCount).toBeGreaterThanOrEqual(5)
    // Weights must sum to 1.0 (renormalization applied across 5 LIKEs)
    expect(record!.professionWeight + record!.skillsWeight + record!.locationWeight +
           record!.modalityWeight + record!.salaryWeight).toBeCloseTo(1.0, 3)
  })

})
```

NOTE on Neo4j in integration tests: The GET /jobs/feed endpoint calls Neo4j's computeFitScore. If Neo4j is not running in the test environment (as in Phase 17 — 26 pre-existing failures), the feed request itself may fail. The third test is written to assert on DB state (likeCount and weight sum) rather than the feed HTTP response, to keep the test robust regardless of Neo4j availability. This is the same pragmatic approach as existing swipe integration tests.
